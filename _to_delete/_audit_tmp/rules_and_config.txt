=========== FILE: firestore.rules ===========
rules_version = '2';

// Chokro — Firestore security rules.
//
// M2 revision. The M1 blocks for `users` and `sellerApplications` are unchanged.
// The `wallets` block has been tightened, and `transactions`, `disposals`,
// `bins`, `config`, `lockouts` and `claimQuotas` are new.
//
// THE ONE RULE THAT MATTERS
// -------------------------
// No client writes a balance. Not a user, not an administrator.
//
// Every field that decides or records a payout — wallet balances, ledger
// entries, disposal status, points awarded, screening results, bin geofences,
// and the points policy itself — is written by the trusted Node service using
// the Firebase Admin SDK, which bypasses these rules entirely. What follows is
// therefore not "the rules that award points"; it is the rules that make it
// impossible for anything except the server to award points.
//
// Why the admin is blocked too: if administrators could write wallets directly,
// there would be two code paths that credit a balance and rules could only
// police one of them. The admin's approve button calls the server like
// everything else. That is what makes the sentence above true without
// qualification.
//
// Corollary: read access is generous here, write access is nearly absent. That
// asymmetry is the design, not an oversight.

service cloud.firestore {
  match /databases/{database}/documents {

    // -------------------------------------------------------------------------
    // Helpers (M1, unchanged)
    // -------------------------------------------------------------------------

    function isSignedIn() {
      return request.auth != null;
    }

    function isSelf(uid) {
      return isSignedIn() && request.auth.uid == uid;
    }

    function role() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role;
    }

    function isAdmin() {
      return isSignedIn() && role() == 'admin';
    }

    function isSeller() {
      return isSignedIn() && (role() == 'seller' || role() == 'admin');
    }

    function userData() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid)).data;
    }

    // Whether this account may act (F5.3).
    //
    // A suspension is either indefinite or timed. Nothing in this system runs
    // on a schedule — Cloud Functions need a billing card and the free Render
    // instance sleeps — so no process exists that could flip `status` back to
    // 'active' when a timed suspension runs out. The expiry is therefore
    // resolved here, at request time, exactly as lockoutActive() resolves its
    // own window a few lines below.
    //
    // A missing `suspendedUntil` means the suspension is indefinite, so its
    // absence must never read as "expired" — hence the explicit key check
    // rather than a comparison against a null.
    //
    // Mirrored by UserModel.isActiveAt, and proven by
    // rules_test/suspension.rules.test.js.
    function isActiveWith(u) {
      return u.status == 'active'
        || (
          u.status == 'suspended'
          && 'suspendedUntil' in u
          && u.suspendedUntil < request.time
        );
    }

    // Reads the user document exactly once.
    //
    // Firestore does not cache get() across calls within one rule evaluation,
    // and each counts against a hard per-request budget of 10 document reads.
    // The previous form called userData() four times, which pushed the disposal
    // create path over the limit and surfaced as an evaluation error rather
    // than a clean denial.
    function isActive() {
      return isSignedIn() && isActiveWith(userData());
    }

    // -------------------------------------------------------------------------
    // Helpers (M2)
    // -------------------------------------------------------------------------

    // A bin that exists and is currently accepting submissions.
    function binIsOpen(binId) {
      return exists(/databases/$(database)/documents/bins/$(binId))
        && get(/databases/$(database)/documents/bins/$(binId)).data.active == true;
    }

    // Whether this user is inside the lockout window for this bin (§7.3, F2.6).
    //
    // Rules cannot run a time-window query across a collection, which is why the
    // lockout is a single document keyed {userId}_{binId} whose expiry can be
    // read in one lookup. Absent document or past expiry means free to submit.
    function lockoutActive(uid, binId) {
      return exists(/databases/$(database)/documents/lockouts/$(uid + '_' + binId))
        && get(/databases/$(database)/documents/lockouts/$(uid + '_' + binId)).data.expiresAt > request.time;
    }

    // -------------------------------------------------------------------------
    // users (M1, unchanged)
    // -------------------------------------------------------------------------

    match /users/{uid} {
      allow read: if isSelf(uid) || isAdmin();

      allow create: if isSelf(uid)
        && request.resource.data.keys().hasAll(['name', 'email', 'role', 'status', 'createdAt'])
        && request.resource.data.role == 'buyer'
        && request.resource.data.status == 'active';

      allow update: if (isSelf(uid)
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['name']))
        || (isAdmin()
          && request.resource.data.diff(resource.data).affectedKeys()
               .hasOnly(['role', 'status', 'suspendedAt', 'reinstatedAt', 'suspendedUntil']));

      allow delete: if false;
    }

    // -------------------------------------------------------------------------
    // wallets — read your own, never write one
    // -------------------------------------------------------------------------

    match /wallets/{uid} {
      allow read: if isSelf(uid) || isAdmin();

      // Creation stays client-side and stays pinned to zero.
      //
      // This is the one client write to a wallet document, and it is deliberate:
      // registration opens the user document and the wallet atomically in a
      // single batch (§6 invariant, and a hard-won lesson from the sibling
      // project). A wallet that can only ever be born holding zero cannot be
      // used to steal anything, so allowing it costs nothing and keeps signup
      // independent of the server being reachable.
      allow create: if isSelf(uid)
        && request.resource.data.balance == 0
        && request.resource.data.keys().hasAll(['userId', 'balance', 'updatedAt'])
        && request.resource.data.userId == uid;

      // Every subsequent balance change belongs to the Admin SDK. An admin is
      // refused here exactly as a buyer is.
      allow update: if false;
      allow delete: if false;
    }

    // -------------------------------------------------------------------------
    // transactions — the ledger. Append-only, and only by the server.
    // -------------------------------------------------------------------------

    match /transactions/{txId} {
      allow read: if isSignedIn()
        && (resource.data.userId == request.auth.uid || isAdmin());

      // NFR-4: every balance change is a ledger entry, and balance is
      // reconstructable from history. Both properties die the moment a client
      // can write, amend or remove an entry.
      allow create, update, delete: if false;
    }

    // -------------------------------------------------------------------------
    // bins — read to scan, server to register
    // -------------------------------------------------------------------------

    match /bins/{binId} {
      // Any signed-in user may resolve a scanned code. Inactive bins are
      // readable too so the app can say "this bin is no longer in service"
      // rather than failing as if the code were fake.
      allow read: if isSignedIn();

      // Written by the server, including for administrators. A bin's
      // coordinates and radius are inputs the server trusts when deciding a
      // payout, so they sit on the server side of the boundary with everything
      // else the server trusts.
      allow create, update, delete: if false;
    }

    // -------------------------------------------------------------------------
    // disposals — the client may submit, and may do nothing else
    // -------------------------------------------------------------------------

    match /disposals/{disposalId} {
      allow read: if isSignedIn()
        && (resource.data.userId == request.auth.uid || isAdmin());

      allow create: if isSignedIn()
        && isActive()
        && request.resource.data.userId == request.auth.uid

        // hasOnly, not hasAll. An exact key set is what keeps `pointsAwarded`,
        // `photoHash`, `screenConfidence`, `reviewedBy` and every other
        // server-owned field out of a client-authored document. hasAll would
        // permit extra keys and the whole guarantee would leak.
        && request.resource.data.keys().hasOnly([
             'userId', 'binId', 'photoUrl', 'photoPublicId',
             'capturedLat', 'capturedLng', 'distanceMeters',
             'declaredItemCount', 'itemType',
             'status', 'flags', 'createdAt'
           ])
        && request.resource.data.keys().hasAll([
             'userId', 'binId', 'photoUrl', 'photoPublicId',
             'capturedLat', 'capturedLng', 'distanceMeters',
             'declaredItemCount', 'itemType',
             'status', 'flags', 'createdAt'
           ])

        // A submission is born pending and unflagged. It cannot arrive approved.
        && request.resource.data.status == 'pending'
        && request.resource.data.flags.size() == 0

        && request.resource.data.declaredItemCount is int
        && request.resource.data.declaredItemCount > 0
        && request.resource.data.declaredItemCount <= 100

        && request.resource.data.capturedLat is number
        && request.resource.data.capturedLng is number
        && request.resource.data.distanceMeters is number

        // Server timestamp only. A client-authored createdAt would undermine
        // the lockout window it is compared against (§7.4).
        && request.resource.data.createdAt == request.time

        && binIsOpen(request.resource.data.binId)
        && !lockoutActive(request.auth.uid, request.resource.data.binId);

      // Screening, approval, rejection and the points award are all server
      // writes. There is no client transition out of pending.
      allow update, delete: if false;
    }

    // -------------------------------------------------------------------------
    // claims — the weakest earn route, and shaped accordingly (F6.1-F6.4)
    // -------------------------------------------------------------------------
    //
    // Structurally identical to `disposals`: the client may create one pending
    // record and may do nothing else. What is missing is the geofence and the
    // lockout, because a self-reported action has nothing objective to check
    // against — no bin, no radius, no distance.
    //
    // The safeguards that remain are the closed action vocabulary enforced
    // below, the weekly quota in `claimQuotas` (server-written, so a client
    // cannot lift its own limit), an administrator's own eyes, and an award
    // that §7.3 validates as strictly lower than a disposal's.
    //
    // There is no auto-approval path here at all. Every decision is a person's.

    match /claims/{claimId} {
      allow read: if isSignedIn()
        && (resource.data.userId == request.auth.uid || isAdmin());

      allow create: if isSignedIn()
        && isActive()
        && request.resource.data.userId == request.auth.uid

        // hasOnly, not hasAll. An exact key set is what keeps `pointsAwarded`,
        // `photoHash` and `reviewedBy` out of a client-authored document.
        && request.resource.data.keys().hasOnly([
             'userId', 'actionType', 'photoUrl', 'photoPublicId',
             'status', 'createdAt'
           ])
        && request.resource.data.keys().hasAll([
             'userId', 'actionType', 'photoUrl', 'photoPublicId',
             'status', 'createdAt'
           ])

        // Born pending. It cannot arrive approved.
        && request.resource.data.status == 'pending'

        // The vocabulary is closed here as well as in the client, because the
        // client's copy is a convenience and this one is the enforcement.
        && request.resource.data.actionType in [
             'treePlanting', 'composting', 'refusedSingleUsePlastic',
             'reusableBagOrBottle', 'communityCleanup'
           ]

        && request.resource.data.photoUrl is string
        && request.resource.data.photoUrl.size() > 0

        // A client clock value is rejected outright (§7.4).
        && request.resource.data.createdAt == request.time;

      // Approval is a payout decision, so it goes to the server like every
      // other one. This denies an administrator too — proven by the test named
      // exactly that in rules_test/claims.rules.test.js.
      allow update, delete: if false;
    }

    // -------------------------------------------------------------------------
    // lockouts and claimQuotas — visible to the user, written by the server
    // -------------------------------------------------------------------------

    match /lockouts/{lockoutId} {
      // Readable so the app can say "you can submit at this bin again in 4h"
      // instead of letting the user photograph a bag for nothing.
      allow read: if isSignedIn();
      allow create, update, delete: if false;
    }

    match /claimQuotas/{quotaId} {
      allow read: if isSignedIn();
      allow create, update, delete: if false;
    }

    // -------------------------------------------------------------------------
    // config — the points policy
    // -------------------------------------------------------------------------

    match /config/{docId} {
      // Every client reads the policy: the checkout screen needs the redemption
      // rate, the submission screen needs the disposal award to show what is
      // pending. None of it is secret.
      allow read: if isSignedIn();

      // Written by the server after PointsPolicy.validate() passes. Rules cannot
      // express the invariant that matters here — that the claim award stays
      // below the disposal award, so the weaker verification route keeps paying
      // less — so the write goes where that check can actually run.
      allow create, update, delete: if false;
    }

    // -------------------------------------------------------------------------
    // sellerApplications (M1, unchanged)
    // -------------------------------------------------------------------------

    match /sellerApplications/{appId} {
      allow read: if isSignedIn()
        && (resource.data.userId == request.auth.uid || isAdmin());

      allow create: if isSignedIn()
        && isActive()
        && request.resource.data.userId == request.auth.uid
        && request.resource.data.status == 'pending'
        && request.resource.data.keys().hasAll([
             'userId', 'businessName', 'description', 'status', 'createdAt'
           ]);

      allow update: if isAdmin()
        && request.resource.data.diff(resource.data).affectedKeys()
             .hasOnly(['status', 'reviewedBy', 'reviewedAt', 'reason']);

      allow delete: if false;
    }

    // -------------------------------------------------------------------------
    // Everything else is denied.
    // -------------------------------------------------------------------------

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
